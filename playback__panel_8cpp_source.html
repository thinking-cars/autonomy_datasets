39PlaybackPanel::~PlaybackPanel() {
40 // Drops the pending request together with its callback, which would otherwise report into a
41 // panel that no longer exists. RViz spins its node from the GUI thread that also destroys the
42 // panel, so no response can be in the middle of being delivered while this runs.
43 client_.reset();
44}

158void PlaybackPanel::showElided(QLabel* label, const QString& text) {
159 const QString shown = label->fontMetrics().elidedText(text, Qt::ElideRight, label->contentsRect().width());
160 if (shown != label->text()) {
161 // Guards against a resize that the new text triggers in turn
162 label->setText(shown);
163 }
164 // Marked up as rich text, so that a long text is wrapped over several lines of the tooltip
165 label->setToolTip(text.isEmpty() ? QString() : QString("<qt>%1</qt>").arg(text.toHtmlEscaped()));
166}

196void PlaybackPanel::createClient() {
197 const QString service_name = service_name_edit_->text().trimmed();
198 if (node_ == nullptr || service_name.isEmpty()) {
199 client_.reset();
200 updateAvailability();
201 return;
202 }
203 client_ = node_->create_client<RequestSamples>(service_name.toStdString());
204 service_available_ = false;
205 updateAvailability();
206}

213void PlaybackPanel::updateAvailability() {
214 const bool available = client_ != nullptr && client_->service_is_ready();
215 if (available == service_available_) {
216 updateControlState();
217 return;
218 }
219 service_available_ = available;
220
221 if (!available && request_in_flight_) {
222 // The node answering the request is gone, so its response will never arrive
223 request_in_flight_ = false;
224 showStatus(tr("The dataset node disappeared while it was processing the request"));
225 } else if (!available) {
226 showStatus(tr("Waiting for service '%1'").arg(service_name_edit_->text().trimmed()));
227 } else if (!request_in_flight_) {
228 showStatus(tr("Connected to '%1'").arg(service_name_edit_->text().trimmed()));
229 }
230 updateControlState();
231}

255void PlaybackPanel::onSkipClicked() {
256 QString error;
257 const std::vector<uint64_t> sample_ids = parseSampleIds(&error);
258 if (!error.isEmpty()) {
259 showStatus(error);
260 return;
261 }
262 if (sample_ids.empty()) {
263 showStatus(tr("Enter the ID of the sample to skip to"));
264 return;
265 }
266 sendRequest(RequestSamples::Request::MODE_SAMPLE_IDS, 0, sample_ids,
267 tr("Skipping ahead to %1 sample(s)...").arg(static_cast<qulonglong>(sample_ids.size())));
268}

274std::vector<uint64_t> PlaybackPanel::parseSampleIds(QString* error) const {
275 std::vector<uint64_t> sample_ids;
276 const QStringList entries = sample_ids_edit_->text().split(QRegularExpression("[,;\\s]+"), Qt::SkipEmptyParts);
277 for (const QString& entry : entries) {
278 bool valid = false;
279 const qulonglong sample_id = entry.toULongLong(&valid);
280 if (!valid) {
281 *error = tr("'%1' is not a sample ID").arg(entry);
282 return {};
283 }
284 sample_ids.push_back(static_cast<uint64_t>(sample_id));
285 }
286 return sample_ids;
287}

289void PlaybackPanel::sendRequest(uint8_t mode,
290 uint64_t num_samples,
291 const std::vector<uint64_t>& sample_ids,
292 const QString& pending_status) {
293 if (client_ == nullptr || !client_->service_is_ready()) {
294 showStatus(tr("Service '%1' is not available").arg(service_name_edit_->text().trimmed()));
295 return;
296 }
297
298 auto request = std::make_shared<RequestSamples::Request>();
299 request->mode = mode;
300 request->num_samples = num_samples;
301 request->sample_ids = sample_ids;
302
303 request_in_flight_ = true;
304 showStatus(pending_status);
305 updateControlState();
306
307 client_->async_send_request(request, [this](rclcpp::Client<RequestSamples>::SharedFuture future) {
308 const auto response = future.get();
309 QString status = QString::fromStdString(response->message);
310 if (response->end_of_dataset) {
311 status += tr("; the dataset has ended");
312 }
313 Q_EMIT requestFinished(describePosition(*response), status, response->success);
314 });
315}

317QString PlaybackPanel::describePosition(const RequestSamples::Response& response) {
318 if (response.published_sample_ids.empty()) {
319 return {};
320 }
321 const uint64_t sample_id = response.published_sample_ids.back();
322 if (response.published_scene_ids.size() != response.published_sample_ids.size()) {
323 return tr("Sample %1").arg(static_cast<qulonglong>(sample_id));
324 }
325 return tr("Sample %1 of scene '%2'")
326 .arg(static_cast<qulonglong>(sample_id))
327 .arg(QString::fromStdString(response.published_scene_ids.back()));
328}

330void PlaybackPanel::applyRequestResult(const QString& position, const QString& status, bool success) {
331 request_in_flight_ = false;
332 if (!position.isEmpty()) {
333 showPosition(position);
334 }
335 showStatus(success ? tr("OK - %1").arg(status) : tr("Failed - %1").arg(status));
336 updateControlState();
337}
